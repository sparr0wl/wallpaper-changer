cmake_minimum_required(VERSION 3.21)
project(WallpaperChanger VERSION 7.41.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(WALLPAPER_RELEASE "7.41f+Trove_of_Terror" CACHE STRING "GitHub release tag")
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia Network)

add_executable(wallpaper-changer WIN32 native/main.cpp)
target_link_libraries(wallpaper-changer PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Network)
target_compile_definitions(wallpaper-changer PRIVATE WALLPAPER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" WALLPAPER_RELEASE="${WALLPAPER_RELEASE}")
qt_add_resources(wallpaper-changer ui PREFIX "/ui" FILES qml/shell.qml qml/ActionButton.qml)
install(TARGETS wallpaper-changer RUNTIME DESTINATION bin)
if(NOT WIN32)
    install(DIRECTORY dota_background DESTINATION lib/wallpaper-changer
            FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
    install(FILES packaging/wallpaper-changer.desktop DESTINATION share/applications)
    install(FILES packaging/wallpaper-changer.svg DESTINATION share/icons/hicolor/scalable/apps)
    install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION share/doc/wallpaper-changer)
endif()

set(CPACK_PACKAGE_NAME "wallpaper-changer")
set(CPACK_PACKAGE_VENDOR "sparr0wl")
set(CPACK_PACKAGE_CONTACT "https://github.com/sparr0wl/wallpaper-changer/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Video menu backgrounds and fonts for Dota 2")
set(CPACK_PACKAGE_VERSION "7.41.0")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_FILE_NAME "wallpaper-changer-${WALLPAPER_RELEASE}-linux-x86_64")
set(CPACK_DEBIAN_PACKAGE_VERSION "7.41.0+trove.of.terror")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "amd64")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "sparr0wl")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.10), liblz4-1, ffmpeg, libqt6core6 (>= 6.4), libqt6gui6, libqt6network6, libqt6qml6, libqt6quick6, libqt6multimedia6, gstreamer1.0-plugins-base, gstreamer1.0-plugins-good, gstreamer1.0-libav, qt6-qpa-plugins, qt6-wayland, qml6-module-qtqml, qml6-module-qtqml-models, qml6-module-qtquick, qml6-module-qtquick-window, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-dialogs, qml6-module-qtquick-templates, qml6-module-qtqml-workerscript, qml6-module-qtmultimedia")
set(CPACK_RPM_PACKAGE_VERSION "7.41f")
set(CPACK_RPM_PACKAGE_RELEASE "1.trove_of_terror")
set(CPACK_RPM_PACKAGE_ARCHITECTURE "x86_64")
set(CPACK_RPM_PACKAGE_LICENSE "LicenseRef-unspecified")
set(CPACK_RPM_PACKAGE_REQUIRES "python3 >= 3.10, glibc >= 2.36, lz4-libs, ffmpeg-free, qt6-qtbase >= 6.4, qt6-qtdeclarative >= 6.4, qt6-qtmultimedia >= 6.4, qt6-qtwayland >= 6.4")
set(CPACK_RPM_PACKAGE_AUTOREQPROV OFF)
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION "/usr" "/usr/bin" "/usr/lib" "/usr/share" "/usr/share/applications" "/usr/share/icons" "/usr/share/icons/hicolor" "/usr/share/icons/hicolor/scalable" "/usr/share/icons/hicolor/scalable/apps" "/usr/share/doc")
include(CPack)
